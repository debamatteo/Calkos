Sub NormalizzaBlindata()
' *** MODIFICA PARAMETRIZZATA 13/05/2026 ***
    Dim pwdInserita As String, ws As Worksheet, rigaFineCell As Range
    Dim lastRow As Long, vElementi As Variant, elemento As Variant
    Dim colLettera As String, col As Long, i As Long
    Dim formula As String, backupPath As String
    Dim fmt As String 'SYSTEMP 11/05/2026 - formato originale cella

    ' --- CONTROLLO PASSWORD ---
    pwdInserita = InputBox("Inserire password per procedere:", "Accesso Protetto")
    If pwdInserita <> PWD_RICHIESTA Then
        If pwdInserita <> "" Then MsgBox "Password errata!", vbCritical
        Exit Sub
    End If

    ' --- BACKUP AUTOMATICO ---
    Set ws = ActiveSheet
    If ThisWorkbook.Path <> "" Then
        backupPath = ThisWorkbook.Path & "\" & Format(Now, "YYYYMMDD_HHMMSS") & "_BACKUP_" & ThisWorkbook.Name
        ThisWorkbook.SaveCopyAs backupPath
    End If

    ' --- RICERCA FINE TABELLA ---
    Set rigaFineCell = ws.Cells.Find(What:="FINE_TABELLA", LookIn:=xlValues, LookAt:=xlWhole)
    If rigaFineCell Is Nothing Then
        MsgBox "Errore: 'FINE_TABELLA' non trovata nel foglio!", vbCritical
        Exit Sub
    End If
    lastRow = rigaFineCell.Row - 1

    ' --- ELABORAZIONE COLONNE ---
    vElementi = Split(OttieniMappatura(), ",")

    For Each elemento In vElementi
        colLettera = Trim(elemento)
        
        If colLettera <> "" Then
            col = ws.Range(colLettera & "1").Column

            For i = RIGA_DATI To lastRow
                With ws.Cells(i, col)

                    ' *** SYSTEMP 11/05/2026 ***
                    ' Salvo il formato originale della cella (mantiene €, %, formati personalizzati)
                    fmt = .NumberFormat

                    ' 1. PARACADUTE: Se la cella contiene un errore (es. #N/D), la salta
                    If Not IsError(.Value) Then
                        
                        ' 2. Processa solo se la cella non è vuota
                        If .Value <> "" Then
                            
                            If .HasFormula Then
                                ' CASO A: Formula
                                formula = Mid(.formula, 2)
                                If InStr(1, formula, "ROUND(", vbTextCompare) = 0 And _
                                   InStr(1, formula, "ARROTONDA(", vbTextCompare) = 0 Then
                                    .formula = "=ROUND(" & formula & "," & DECIMALI & ")"
                                End If
                            
                            ElseIf IsNumeric(.Value) Then
                                ' CASO B: Numero manuale
                                .Value = WorksheetFunction.Round(.Value, DECIMALI)
                            End If
                            
                        End If
                    End If


' SOLUZIONE 1 ' Ripristino del formato originale (mantiene € solo dove già c’era)
'                    ' *** SYSTEMP 11/05/2026 ***
'                    ' Ripristino del formato originale (mantiene € solo dove già c’era)
'                    .NumberFormat = fmt




'' SOLUZIONE 2
'                    ' *** MODIFICA PARAMETRIZZATA 13/05/2026 ***
'                    ' Invece di ripristinare il formato vecchio (fmt),
'                    ' costruiamo il formato basandoci sulla costante DECIMALI_VISUALE
'
'                    Dim stringaFormato As String
'                    stringaFormato = "0." & String(DECIMALI_VISUALE, "0")
'
'                    ' Se la cella originale aveva il simbolo €, lo manteniamo
'                    If InStr(1, fmt, "€") > 0 Then
'                        .NumberFormat = "#,##0." & String(DECIMALI_VISUALE, "0") & " €"
'                    Else
'                        .NumberFormat = stringaFormato
'                    End If
                    
                    
' ' SOLUZIONE 3  (€ 1,50->1,5000 €)
'                Dim stringaFormato As String
'                stringaFormato = "#,##0." & String(DECIMALI_VISUALE, "0")
'
'                If InStr(1, fmt, "€", vbTextCompare) > 0 Or _
'                InStr(1, fmt, "[$€", vbTextCompare) > 0 Then
'
'                .NumberFormat = stringaFormato & " €"
'
'                Else
'
'                .NumberFormat = stringaFormato
'
'                End If
'
'


                'SOLUZIONE 4 € 1,50 --> € 1,5000
                
                Dim nuovoFormato As String
                
                nuovoFormato = fmt
                
                ' Se esiste una parte decimale nel formato
                If InStr(nuovoFormato, ".") > 0 Then
                    
                    Dim parteIntera As String
                    Dim parteDecimale As String
                    Dim posPunto As Long
                    Dim j As Long
                    
                    posPunto = InStr(nuovoFormato, ".")
                    
                    parteIntera = Left(nuovoFormato, posPunto)
                    
                    parteDecimale = ""
                    
                    ' Ricostruisce eventuali suffissi tipo €, %, ecc.
                    For j = posPunto + 1 To Len(nuovoFormato)
                        
                        Dim ch As String
                        ch = Mid(nuovoFormato, j, 1)
                        
                        If ch Like "[0#]" Then
                            ' salta i vecchi decimali
                        Else
                            parteDecimale = Mid(nuovoFormato, j)
                            Exit For
                        End If
                        
                    Next j
                    
                    nuovoFormato = parteIntera & String(DECIMALI_VISUALE, "0") & parteDecimale
                
                Else
                    
                    ' Se non aveva decimali li aggiunge mantenendo il resto
                    nuovoFormato = nuovoFormato & "." & String(DECIMALI_VISUALE, "0")
                
                End If
                
                .NumberFormat = nuovoFormato

                    


                End With
            Next i

        End If
    Next elemento

    MsgBox "Operazione conclusa con successo!" & vbCrLf & _
           "Formule e valori manuali sono stati normalizzati.", vbInformation, "Fine"
End Sub
