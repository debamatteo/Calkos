// In tutti gli altri moduli, la funzione controlla() diventa:


/** MODIFICA 11/05/2026
 * Valuta se il valore nel database diverge dal calcolo teorico della formula.
 * Viene utilizzata nei moduli (Cobral, SystemCore, ecc.) per inizializzare lo stato dei lucchetti.
 */
function controlla(idHidden, idBtn, idDisplay, fnCalcola, setLock) {
    // Si utilizza una precisione a 3 decimali per il confronto.
    // Questo risolve le micro-discrepanze (es. 194.428 vs 194.4276) causate dai diversi 
    // arrotondamenti nei record storici importati da Excel, che non sono vere forzature.
    const DECIMALI_CONFRONTO = 3;
    
    const valoreDB   = safeField(idHidden); // Recupera il valore originale dal DB (campo hidden)
    const valoreCalc = fnCalcola();         // Esegue la formula JS per ottenere il valore atteso
    
    // Normalizzazione: arrotonda entrambi i valori alla stessa precisione (3 decimali)
    // trasformando il risultato in Number per il calcolo matematico.
    const dbArr      = Number(valoreDB.toFixed(DECIMALI_CONFRONTO));
    const calcArr    = Number(valoreCalc.toFixed(DECIMALI_CONFRONTO));

    // Se la differenza tra i valori normalizzati supera la SOGLIA (0.0001),
    // significa che il valore è stato effettivamente forzato a mano dall'utente.
    if (Math.abs(dbArr - calcArr) > SOGLIA) {
        log("Valore forzato rilevato per", idHidden,
            "| DB =", valoreDB, "(→" + dbArr + ")",
            "| Calcolato =", valoreCalc, "(→" + calcArr + ")",
            "→ lucchetto aperto automaticamente");
        
        // Sblocca la logica (setLock false = non più vincolato alla formula)
        setLock(false);
        
        // Aggiorna visivamente il lucchetto e rende il campo editabile
        aggiornaUILucchetto(idBtn, idDisplay, false);
    }
}

function apriSeNecessario(chiave, idBtn, idDisplay, valoreDB, valoreCalc) {
    /**
     * Arrotonda entrambi i valori a 3 decimali prima del confronto.
     * Motivazione: i record storici importati da Excel sono stati salvati
     * con precisione variabile (alcuni a 3 decimali, altri a 5).
     * La query di verifica sul DB ha confermato che NON esistono divergenze
     * reali superiori a 0.001, quindi qualsiasi differenza sotto quella soglia
     * è esclusivamente un artefatto di arrotondamento storico, non una
     * forzatura intenzionale dell'utente.
     * Soglia 0.001 = 1 centesimo di millesimo → nessun impatto commerciale.
     */
    const DECIMALI_CONFRONTO = 3;
    const dbArrotondato   = Number(valoreDB.toFixed(DECIMALI_CONFRONTO));
    const calcArrotondato = Number(valoreCalc.toFixed(DECIMALI_CONFRONTO));

    if (Math.abs(dbArrotondato - calcArrotondato) > SOGLIA) {
        log("⚠ Valore forzato rilevato per [" + chiave + "]" +
            " | DB=" + valoreDB + " (→" + dbArrotondato + ")" +
            " | Formula=" + valoreCalc + " (→" + calcArrotondato + ")" +
            " → lucchetto aperto automaticamente");
        manuale[chiave] = true;
        $(idDisplay).prop("readonly", false).removeClass("input-readonly");
        $(idBtn).removeClass("btn-success").addClass("btn-danger")
                .html('<i class="bi bi-unlock-fill"></i>');
    }
}




VERIFICA ANOMALIE 
Cosa fare praticamente: lancia questa query per capire quanti record hanno davvero 
divergenze sostanziali vs divergenze da arrotondamento:
sqlSELECT NumeroOrdine, NumeroDDT,
       TotaleProvvigioneDaFatturare,
       TotaleSommaProvvigioni,
       PrezzoVendita,
       ABS(TotaleProvvigioneDaFatturare - 
           CASE WHEN TotaleSommaProvvigioni > PrezzoVendita 
                THEN TotaleSommaProvvigioni 
                ELSE PrezzoVendita END) AS Divergenza
FROM ProspettoElektraWire
WHERE ABS(TotaleProvvigioneDaFatturare - 
          CASE WHEN TotaleSommaProvvigioni > PrezzoVendita 
               THEN TotaleSommaProvvigioni 
               ELSE PrezzoVendita END) > 0.001
ORDER BY Divergenza DESC
Se i risultati hanno divergenze piccole (< 0.01) sono tutti arrotondamenti storici — 
puoi alzare la soglia a 0.005 o 0.01 in rilevaDivergenze()
 e il problema sparisce. 
 Se ci sono divergenze grandi (> 0.1) 
 quelli sono veri valori forzati che il lucchetto deve proteggere.
